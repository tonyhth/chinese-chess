## 详细实施

### B.1 重写 EmbeddedPikafishEngine.swift

**核心改动**：
- `import Pikafish`（modulemap），直接调用 C 函数
- 删除 `import PikafishEngine`（ObjC Wrapper）
- 删除 `#if os(iOS)` 包裹整个文件
- `bestMove` 在 `DispatchQueue.global()` 执行阻塞调用
- buffer 大小 32 字节

```swift
import Foundation
import Pikafish  // 通过 modulemap 导入 C API（iOS + macOS 统一）

actor EmbeddedPikafishEngine: ChessEngine {
    nonisolated let displayName: String = "Pikafish"
    nonisolated let engineType: EngineType = .external
    private(set) var isReady = false
    private var cachedVersion: String = "unknown"

    // MARK: - Lifecycle

    func start() async throws {
        guard !isReady else { return }
        pikafish_init()

        // 缓存引擎版本（start 后 C API 数据已就绪）
        if let v = pikafish_get_version() {
            cachedVersion = String(cString: v)
        }

        configureTTSize()
        isReady = true
    }

    func shutdown() {
        guard isReady else { return }
        pikafish_quit()
        isReady = false
    }

    deinit {
        if isReady {
            pikafish_quit()
        }
    }

    // MARK: - ChessEngine

    func bestMove(fen: String, moves: [String], difficulty: AIDifficulty) async -> String {
        guard isReady else { return "0000" }
        let depth = mapDifficultyToDepth(difficulty)
        let movesStr = moves.joined(separator: " ")

        // 在 DispatchQueue.global() 执行阻塞调用，避免占用 cooperative pool
        return await withCheckedThrowingContinuation { continuation in
            DispatchQueue.global(qos: .userInitiated).async { [self] in
                var buffer = [CChar](repeating: 0, count: 32)  // 32 bytes，足够容纳 UCI 走法
                let result = fen.withCString { fenCStr in
                    movesStr.withCString { movesCStr in
                        pikafish_best_move(
                            fenCStr, movesCStr,
                            Int32(depth), 0,
                            &buffer, Int32(buffer.count)
                        )
                    }
                }

                switch result {
                case 0:
                    // 正常完成
                    continuation.resume(returning: String(cString: buffer))
                case -1:
                    // 搜索被中断，返回截至中断时的最佳走法
                    let move = String(cString: buffer)
                    continuation.resume(returning: move.isEmpty ? "0000" : move)
                default:
                    // 参数错误或其他失败
                    continuation.resume(returning: "0000")
                }
            }
        }
    }

    /// 立即停止搜索（原子操作，线程安全，可从任意线程调用）
    nonisolated func stopSearchImmediate() {
        pikafish_stop()
    }

    func stopSearch() {
        pikafish_stop()
    }

    func newGame() {
        pikafish_new_game()
    }

    var version: String { cachedVersion }

    // MARK: - Configuration

    private func configureTTSize() {
        let physicalMemory = ProcessInfo.processInfo.physicalMemory
        let ttSizeMB: Int

        #if os(iOS)
        // iOS：保守策略（后台内存压力更大）
        if physicalMemory < 2_000_000_000 {
            ttSizeMB = 16
        } else if physicalMemory >= 4_000_000_000 {
            ttSizeMB = 64
        } else {
            ttSizeMB = 32
        }
        #else
        // macOS：基于物理内存的 10%，上限 256MB，下限 64MB
        let dynamicMB = Int(physicalMemory / 10 / 1_000_000)
        ttSizeMB = max(64, min(dynamicMB, 256))
        #endif

        pikafish_set_option("Hash", String(ttSizeMB))
    }

    private func mapDifficultyToDepth(_ difficulty: AIDifficulty) -> Int {
        switch difficulty {
        case .beginner: return 2
        case .easy:     return 5
        case .medium:   return 10
        case .hard:     return 18
        case .master:   return 24
        }
    }
}
```

### B.2 删除 ObjC Wrapper

删除以下文件和 Xcode 引用：
- `AI/PikafishEngine.h`
- `AI/PikafishEngine.m`

> iOS 侧也同步迁移到 modulemap，不再需要 ObjC Wrapper。

### B.3 删除 bridging header 引用

从 `ChineseChess-Bridging-Header.h` 中移除：
```objc
// ❌ 删除
#import "PikafishEngine.h"
```

如果 bridging header 只剩空壳，可考虑删除整个文件（modulemap 不需要 bridging header）。

### B.4 平台条件编译

仅 `configureTTSize()` 使用 `#if os(iOS)` / `#else` 区分内存策略。其他方法（bestMove、newGame 等）平台无关，不需要条件编译。

### B.5 引擎信息缓存

- `displayName`：编译时常量 `"Pikafish"`（nonisolated let），不依赖 C API
- `version`：`start()` 后调用 `pikafish_get_version()` 缓存到 `cachedVersion`

## 风险点

| 风险 | 影响 | 回退 |
|------|------|------|
| Swift C 互操作边界问题（字符串转换） | bestMove 返回错误 | 使用 `withCString` 确保生命周期 |
| `pikafish_get_version()` 返回 NULL | version 为空 | nil 检查 + fallback "unknown" |
| bestMove 在 DispatchQueue.global 执行的 continuation 安全性 | 运行时错误 | `withCheckedThrowingContinuation` 保证 resume 恰好一次 |
| iOS 侧删除 ObjC Wrapper 后编译失败 | 阻塞 | 检查所有引用 PikafishEngine 的代码，确保已迁移 |
| bridging header 删除后其他 ObjC 代码编译失败 | 阻塞 | 仅删除 PikafishEngine 引用，保留 bridging header |
