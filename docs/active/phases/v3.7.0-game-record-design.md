## 三、存储方案升级

### 3.1 从 UserDefaults 迁移到 JSON 文件

**为什么不用 SQLite**：
- 数据量级：个人对弈记录最多几百局，JSON 足够
- 复杂度：SQLite 需要 schema migration，JSON 零迁移成本
- 备份：JSON 文件可直接 iCloud 同步、AirDrop 分享
- 调试：人类可读

**存储路径**：
```
App Documents/
  GameRecords/
    index.json              // id 列表 + 元数据摘要（列表展示用，避免全量加载）
    {uuid}.json             // 单条记录
```

### 3.2 GameRecordStore 替代 GameHistoryStore

```swift
/// 新存储引擎（替代 GameHistoryStore）
final class GameRecordStore {
    static let shared = GameRecordStore()

    private let fileManager = FileManager.default
    private let baseURL: URL          // Documents/GameRecords/
    private let indexURL: URL         // Documents/GameRecords/index.json

    // 内存缓存：列表展示用摘要
    private(set) var summaries: [RecordSummary] = []

    struct RecordSummary: Codable, Identifiable {
        let id: UUID
        let title: String
        let date: Date
        let result: GameState
        let totalMoves: Int
        let difficulty: AIDifficulty
        let source: GameRecord.RecordSource
    }

    // MARK: - 读取

    /// 全部记录摘要（列表展示用，从 index.json 加载）
    func loadSummaries() -> [RecordSummary]

    /// 加载完整记录（按需加载，避免内存占用）
    func loadRecord(id: UUID) -> GameRecord?

    /// 记录数量
    var count: Int { summaries.count }

    // MARK: - 写入（线程安全 + 去重）

    /// 串行写入队列，防止并发写坏文件
    private let writeQueue = DispatchQueue(label: "com.chinesechess.recordstore", qos: .userInitiated)

    /// 添加记录（自动去重）
    func addRecord(_ record: GameRecord) {
        writeQueue.sync {
            // P0 去重：同 id 不重复写入
            guard !summaries.contains(where: { $0.id == record.id }) else { return }

            // 1. 写入单条文件（原子写入：先写临时文件再 rename）
            let fileURL = baseURL.appendingPathComponent("\(record.id.uuidString).json")
            let tmpURL = baseURL.appendingPathComponent("\(record.id.uuidString).tmp")
            do {
                let data = try JSONEncoder().encode(record)
                try data.write(to: tmpURL, options: .atomic)
                try FileManager.default.moveItem(at: tmpURL, to: fileURL)
            } catch {
                // 写入失败，不更新 index
                return
            }

            // 2. 更新内存摘要
            summaries.insert(RecordSummary(from: record), at: 0)

            // 3. 更新 index.json
            persistIndex()
        }
    }

    /// 更新记录（改标题/标签）
    func updateRecord(_ record: GameRecord) {
        writeQueue.sync {
            // 1. 覆写单条文件
            let fileURL = baseURL.appendingPathComponent("\(record.id.uuidString).json")
            guard let data = try? JSONEncoder().encode(record) else { return }
            try? data.write(to: fileURL, options: .atomic)

            // 2. 更新内存摘要
            if let idx = summaries.firstIndex(where: { $0.id == record.id }) {
                summaries[idx] = RecordSummary(from: record)
            }

            // 3. 更新 index.json
            persistIndex()
        }
    }

    /// 删除记录
    func deleteRecord(id: UUID) {
        writeQueue.sync {
            // 1. 删除单条文件
            let fileURL = baseURL.appendingPathComponent("\(id.uuidString).json")
            try? FileManager.default.removeItem(at: fileURL)

            // 2. 更新内存摘要
            summaries.removeAll { $0.id == id }

            // 3. 更新 index.json
            persistIndex()
        }
    }

    /// 清空全部
    func clearAll() {
        writeQueue.sync {
            // 1. 删除所有 .json 文件
            for summary in summaries {
                let fileURL = baseURL.appendingPathComponent("\(summary.id.uuidString).json")
                try? FileManager.default.removeItem(at: fileURL)
            }
            summaries = []
            persistIndex()
        }
    }

    // MARK: - 内部

    /// 持久化 index.json（原子写入）
    private func persistIndex() {
        guard let data = try? JSONEncoder().encode(summaries) else { return }
        let tmpURL = baseURL.appendingPathComponent("index.tmp")
        try? data.write(to: tmpURL, options: .atomic)
        try? FileManager.default.moveItem(at: tmpURL, to: indexURL)
    }
}
```

### 3.3 存储上限与排序

- **不设硬上限**，改为磁盘空间策略：单条记录约 2-5KB，1000 局 ≈ 5MB，可接受
- index.json 只存摘要，加载快
- 完整记录按需加载（点进回放时才读文件）

**默认排序**（P2 修复）：按日期倒序（最新在前）
- index.json 中 summaries 按插入顺序维护（新记录 insert at: 0）
- UI 后续可加排序选项（按日期/按难度/按来源），v3.7.0 先只做默认排序

### 3.4 迁移策略（原子性保障）

App 启动时检测 UserDefaults 中是否有旧数据，一次性迁移到 JSON 文件：

```swift
// DataMigration.swift 新增
static func migrateGameHistoryToFiles() {
    let migrated = UserDefaults.standard.bool(forKey: "chinesechess.history.migrated")
    guard !migrated else { return }

    let oldStore = GameHistoryStore.shared
    guard oldStore.count > 0 else {
        // 无旧数据，直接标记已迁移
        UserDefaults.standard.set(true, forKey: "chinesechess.history.migrated")
        return
    }

    let newStore = GameRecordStore.shared

    // 原子迁移：逐条写入，任何一条失败都中止
    var migratedRecords: [GameRecord] = []
    for record in oldStore.records {
        var r = record
        r.source = .versusAI
        // 先写入文件，确认成功后再继续
        do {
            let data = try JSONEncoder().encode(r)
            let fileURL = newStore.baseURL.appendingPathComponent("\(r.id.uuidString).json")
            try data.write(to: fileURL, options: .atomic)
            migratedRecords.append(r)
        } catch {
            // 单条写入失败 → 中止迁移，保留旧数据
            // 已写入的记录保留（不回滚，因为文件写入是原子的不会半写）
            return
        }
    }

    // 全部写入成功 → 更新 index.json
    newStore.reloadSummaries()

    // 确认 JSON 文件完整后才清空旧存储
    oldStore.clearAll()
    UserDefaults.standard.set(true, forKey: "chinesechess.history.migrated")
}
```

**原子性保障**：
1. 每条记录用 `.atomic` 写入（先写临时文件再 rename），不会出现半写文件
2. 全部写入成功后才清空旧存储——任何一步失败，旧数据完好
3. 迁移标记只在最后设置，中途崩溃下次重试
4. 已写入的文件不回滚（单文件原子写入保证完整性）

## 四、PGN 格式定义

### 4.1 中国象棋 PGN 变体

国际象棋 PGN 标准定义了七段必选标签和可选标签。中国象棋无官方 PGN 标准，参考 xqbase 规范定义变体：

**必选标签**：

| 标签 | 含义 | 示例 |
|------|------|------|
| `[Event "??"]` | 赛事 | `"人机对弈"` / `"残局练习"` |
| `[Site "??"]` | 地点 | `"中国象棋"` |
| `[Date "????.??.??"]` | 日期 | `"2026.06.29"` |
| `[Round "1"]` | 轮次 | `"1"` |
| `[Red "玩家"]` | 红方名称 | `"洪涛"` |
| `[Black "AI-高级"]` | 黑方名称 | `"Pikafish"` |
| `[Result "*"]` | 结果 | `"1-0"` / `"0-1"` / `"1/2-1/2"` / `"*"` |

**可选标签**（中国象棋扩展）：

| 标签 | 含义 | 示例 |
|------|------|------|
| `[FEN "..."]` | 初始 FEN | `"rnbakabnr/9/... w - - 0 1"` |
| `[Format "ICCS"]` | 走法格式 | `"ICCS"`（默认）/ `"Chinese"` |
| `[Difficulty "..."]` | AI 难度 | `"hard"` |
| `[PuzzleId "..."]` | 残局 ID | `"puzzle_001"` |
| `[Source "..."]` | 来源 | `"versusAI"` / `"puzzle"` / `"imported"` |

**结果编码**：
- 红胜：`1-0`
- 黑胜：`0-1`
- 和棋：`1/2-1/2`
- 进行中/未完成：`*`

### 4.2 走法格式与行分隔符

默认使用 ICCS 坐标（`h2e2`），与国际象棋 PGN 的 SAN 类似但更精确：

```
1. h2e2 b9c7 2. i0h0 i9i7 3. b0c2 h9g7
```

**ICCS 坐标规则**：
- 列：a-i（红方从右到左，a=九路，i=一路）
- 行：0-9（0=红方底线，9=黑方底线；Board.row 0→ICCS 9，Board.row 9→ICCS 0）
- 四字符：fromCol + fromRow + toCol + toRow

**中文棋谱格式**（可选 `[Format "Chinese"]`）：
```
1. 炮二平五 马8进7 2. 車一进二 車9平8
```

#### 行分隔符规范（P0 修复）

**严格定义**：走法段所有走法在同一行，用空格分隔。回车换行仅出现在标签段和标签/走法段之间。

编码规则：
1. 走法段是一个连续的单行文本，`1.` `2.` ... 回合号标记每步
2. ICCS 走法严格 4 字符（`[a-i][0-9][a-i][0-9]`），不含空格/逗号/特殊字符
3. 中文棋谱格式暂只支持导出，导入需 ChineseNotationParser（v3.8.0）
4. 导入解析时：忽略所有换行符，将走法段合并为单行后按空格 tokenize
5. 回合号（`1.` `2.` ...）作为分隔标记，不参与映射——走法序号 = 出现顺序

**为什么不用换行分隔**：中文棋谱含逗号（`炮二平五，马8进7`）和多字节字符，换行位置不确定。单行+空格+回合号标记是最可靠的解析策略。

**PGNImporter 走法解析伪代码**：
```swift
// 1. 将走法段所有行合并
let movetext = moveLines.joined(separator: " ")

// 2. 去掉回合号，提取纯走法 token
let tokens = movetext
    .components(separatedBy: .whitespaces)
    .filter { !$0.isEmpty && !($0.hasSuffix(".") && $0.dropLast().allSatisfy(\.isNumber)) }
    // 过滤回合号如 "1." "2." 等
    .filter { !["1-0", "0-1", "1/2-1/2", "*"].contains($0) }
    // 过滤结果标记

// 3. tokens[0] = 第一步，tokens[1] = 第二步，...（严格按出现顺序）
// 4. 每个 token 按 ICCS 格式解析为 Move
```

### 4.4 PGN 导入容错策略（P1 修复）

**策略：单局整体回滚，多局部分成功**

| 场景 | 策略 | 原因 |
|------|------|------|
| 标签缺失 | 默认值填充，记录 warning | 不影响走法解析 |
| FEN 缺失 | 假定标准开局 | 大多数对弈从标准开局开始 |
| Format 缺失 | 默认 ICCS | 与导出一致 |
| 单局中有非法走法 | **整体回滚该局**，记录 warning | 部分走法无意义（缺步无法回放） |
| 多局中某局解析失败 | 跳过该局，继续解析下一局 | 不让一局坏棋谱影响整批 |
| 导入结果 UI | 成功 N 局 + 警告 M 条 | 用户可看到哪些没导入成功 |

**为什么单局非法走法回滚而非跳过**：象棋走法有严格依赖（每步依赖上一步的棋盘状态），跳过中间某步后后续步全部错位，导入的棋谱无法回放，不如不导入。

**导入流程**：
```swift
static func parse(_ pgnText: String) -> ImportResult {
    var records: [GameRecord] = []
    var warnings: [String] = []

    let games = splitGames(pgnText)

    for (i, gameText) in games.enumerated() {
        do {
            let record = try parseSingleGame(gameText)
            records.append(record)
        } catch PGNError.illegalMove(let step, let moveStr) {
            warnings.append("第 \(i + 1) 局第 \(step) 步非法走法(\(moveStr))，已跳过该局")
        } catch {
            warnings.append("第 \(i + 1) 局解析失败：\(error.localizedDescription)")
        }
    }

    return ImportResult(records: records, warnings: warnings)
}
```

### 4.5 PGN 导出器

```swift
/// PGN 导出器
struct PGNExporter {

    /// 单局导出
    static func export(_ record: GameRecord) -> String {
        var lines: [String] = []

        // 必选标签
        lines.append("[Event \"\(eventTitle(record))\"]")
        lines.append("[Site \"中国象棋\"]")
        lines.append("[Date \"\(formatDate(record.date))\"]")
        lines.append("[Round \"1\"]")
        lines.append("[Red \"\(record.redPlayer.name)\"]")
        lines.append("[Black \"\(record.blackPlayer.name)\"]")
        lines.append("[Result \"\(resultString(record.result))\"]")

        // 可选标签
        if let fen = record.initialFEN {
            // P0 修复：归一化后比较，避免 FEN 编码差异误判
            if !FENParser.isStandardInitial(fen) {
                lines.append("[FEN \"\(fen)\"]")
            }
        }
        lines.append("[Format \"ICCS\"]")
        if record.source != .imported {
            lines.append("[Difficulty \"\(record.difficulty.rawValue)\"]")
        }
        if let puzzleId = record.puzzleId {
            lines.append("[PuzzleId \"\(puzzleId)\"]")
        }
        lines.append("[Source \"\(record.source.rawValue)\"]")

        // 空行分隔标签和走法
        lines.append("")

        // 走法（ICCS 格式，每回合一行）
        let iccsMoves = record.moves.map { move -> String in
            let fileChars = Array("abcdefghi")
            let from = String(fileChars[move.from.col]) + String(9 - move.from.row)
            let to = String(fileChars[move.to.col]) + String(9 - move.to.row)
            return from + to
        }

        var moveLines: [String] = []
        var i = 0
        var roundNum = 1
        while i < iccsMoves.count {
            var line = "\(roundNum). \(iccsMoves[i])"
            if i + 1 < iccsMoves.count {
                line += " \(iccsMoves[i + 1])"
            }
            moveLines.append(line)
            i += 2
            roundNum += 1
        }
        // 附加结果
        if let last = moveLines.last {
            moveLines[moveLines.count - 1] = last + " \(resultString(record.result))"
        }

        lines.append(moveLines.joined(separator: " "))
        return lines.joined(separator: "\n")
    }

    /// 批量导出（多局合并）
    static func exportBatch(_ records: [GameRecord]) -> String {
        records.map { export($0) }.joined(separator: "\n\n")
    }

    /// 批量导出（容错版本，P2 修复）
    /// 单条导出失败不阻断整批，用注释替代失败记录
    static func exportBatchSafe(_ records: [GameRecord]) -> (pgn: String, failedCount: Int) {
        var parts: [String] = []
        var failedCount = 0
        for record in records {
            do {
                let pgn = try exportSafe(record)
                parts.append(pgn)
            } catch {
                parts.append("[Event \"导出失败\"]\n[Date \"\(formatDate(record.date))\"]\n% ERROR: \(error.localizedDescription)")
                failedCount += 1
            }
        }
        return (parts.joined(separator: "\n\n"), failedCount)
    }

    // P2: 导入来源标记 — 记录中 source = .imported，difficulty 默认 .medium
    private static func defaultDifficulty(for source: GameRecord.RecordSource) -> AIDifficulty {
        return .medium  // 导入记录无难度信息时默认中等
    }

    // P2: 导入来源标记 — imported 记录的 redPlayer/blackPlayer difficulty 设为 nil
    private static func playerInfoFromTags(_ tags: [String: String], side: String) -> PlayerInfo {
        let name = tags[side] ?? (side == "Red" ? "红方" : "黑方")
        return PlayerInfo(name: name, isAI: false, difficulty: nil)
    }

    // MARK: - 辅助

    private static func eventTitle(_ record: GameRecord) -> String {
        switch record.source {
        case .versusAI: return "人机对弈"
        case .puzzle: return "残局练习"
        case .imported: return "??"
        case .freePlay: return "自由对弈"
        }
    }

    private static func formatDate(_ date: Date) -> String {
        let f = DateFormatter()
        f.dateFormat = "yyyy.MM.dd"
        return f.string(from: date)
    }

    private static func resultString(_ result: GameState) -> String {
        switch result {
        case .redWon: return "1-0"
        case .blackWon: return "0-1"
```

## 五、FEN 精确推算

### 5.1 问题

`AnalysisViewModel.computeFEN(before:)` 当前直接返回 `initialFEN`，无论 index 是多少。这意味着 PositionAnalyzer 分析每步时用的都是初始局面，而非走棋前的实际局面——分析结果完全错误。

### 5.2 FEN 归一化比较

**P0 修复**：FEN 编码不唯一（如 `R1` vs `R1` 同义但不同写法、行走方后字段差异），直接字符串比较不可靠。

在 FENParser 中新增归一化比较方法：

```swift
extension FENParser {
    /// 判断 FEN 是否代表标准开局（归一化后比较）
    static func isStandardInitial(_ fen: String) -> Bool {
        // 方案：解析 FEN → Board → 重新生成 FEN → 与 standardInitial 比较
        // 这样消除了所有编码差异（数字简写、字段顺序、空格等）
        guard let board = parse(fen: fen) else { return false }
        let normalized = generate(board: board)
        return normalized == standardInitial
    }
}
```

**为什么不用字符串归一化**：FEN 变体太多（缺少后半段字段、数字简写不同），直接字符串操作易遗漏。parse→generate 路径是 FENParser 自身代码，保证了格式一致性。

**性能**：只在导出/导入时调用，不在热路径上，可忽略。

### 5.3 方案：FENRebuilder

```swift
/// 从走法序列精确推算任意步骤的 FEN
struct FENRebuilder {

    /// 推算第 index 步走棋前的 FEN
    static func computeFEN(initialFEN: String, moves: [GameMove], before index: Int) -> String {
        let board = Board(fen: initialFEN)
        for i in 0..<index {
            let gm = moves[i]
            let piece = board.piece(at: gm.from) ?? gm.piece
            let captured = board.piece(at: gm.to)
            let move = Move(piece: piece, from: gm.from, to: gm.to, captured: captured)
            board.execute(move)
        }
        return FENParser.generate(board: board)
    }

    /// 批量推算：生成每步走棋前后的 FEN 列表
    /// - Returns: FEN 数组，[0] = initialFEN, [1] = 第一步走后, ...
    static func computeAllFENs(initialFEN: String, moves: [GameMove]) -> [String] {
        var board = Board(fen: initialFEN)
        var fens = [initialFEN]
        for gm in moves {
            let piece = board.piece(at: gm.from) ?? gm.piece
            let captured = board.piece(at: gm.to)
            let move = Move(piece: piece, from: gm.from, to: gm.to, captured: captured)
            board.execute(move)
            fens.append(FENParser.generate(board: board))
        }
        return fens
    }
}
```

### 5.4 FEN 推算链路验证（P1 修复）

从 GameMove 到 Board.execute 再到 FENParser.generate 的完整链路需验证可用性：

**已验证的关键路径**：
1. `GameMove` 包含 `piece/from/to/captured` → 可构建 `Move(piece:from:to:captured:)`
2. `Board.execute(_ move: Move)` 已实现，包含吃子移除 + 棋子移动 + currentTurn 切换
3. `FENParser.generate(board:)` 已实现，按行扫描棋子生成 FEN
4. `Board.piece(at:)` 用于获取目标位置棋子（captured）

**潜在问题**：GameMove 中记录的 `piece` 是走棋前的棋子快照，而 `Board.piece(at: gm.from)` 在回放时可能返回不同的棋子对象（因为 Board 是重建的）。因此 FENRebuilder 中使用 `board.piece(at: gm.from) ?? gm.piece`——先查 Board 当前状态，查不到再 fallback 到记录的 piece。

**验证方法**：Cody 编码时需写单元测试：
```swift
func testFENRoundTrip() {
    let board = Board()  // 标准开局
    let moves = simulateFullGame()  // 模拟一局完整对弈
    let fens = FENRebuilder.computeAllFENs(initialFEN: FENParser.standardInitial, moves: moves)
    // 每一步：Board.execute 后的 FEN == FENRebuilder 推算的 FEN
    var verifyBoard = Board()
    for (i, move) in moves.enumerated() {
        let piece = verifyBoard.piece(at: move.from) ?? move.piece
        let captured = verifyBoard.piece(at: move.to)
        let m = Move(piece: piece, from: move.from, to: move.to, captured: captured)
        verifyBoard.execute(m)
        XCTAssertEqual(FENParser.generate(board: verifyBoard), fens[i + 1])
    }
}
```

### 6.4 PGN 导入解析器

```swift
/// PGN 导入解析器
struct PGNImporter {

    struct ImportResult {
        let records: [GameRecord]
        let warnings: [String]     // 如"第3步非法走法已跳过"
    }

    /// 解析 PGN 文本（支持单局和多局）
    static func parse(_ pgnText: String) -> ImportResult {
        var records: [GameRecord] = []
        var warnings: [String] = []

        let games = splitGames(pgnText)

        for (i, gameText) in games.enumerated() {
            do {
                let record = try parseSingleGame(gameText)
                records.append(record)
            } catch {
                warnings.append("第 \(i + 1) 局解析失败：\(error.localizedDescription)")
            }
        }

        return ImportResult(records: records, warnings: warnings)
    }

    // MARK: - 内部

    /// 按 double newline 或新标签段分割多局 PGN
    private static func splitGames(_ text: String) -> [String]

    /// 解析单局 PGN
    private static func parseSingleGame(_ text: String) throws -> GameRecord

    /// 解析标签段
    private static func parseTags(_ lines: [String]) -> [String: String]

    /// 解析走法段（ICCS 格式）
    private static func parseMoves(_ text: String, initialFEN: String) throws -> [GameMove]
}
```
