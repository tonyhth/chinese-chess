## 一、现状分析

| 方式 | 调用位置 | 覆盖成就数 |
|------|----------|-----------|
| `checkAndUnlock(profile:)` 统计检查 | `DataMigration.swift`（迁移时调用一次） | 16 个 |
| `unlock(id)` 硬编码调用 | `TutorialViewModel.swift`（教程完成时） | 2 个（与 checkAndUnlock 重叠） |

**结果：34 个成就中 18 个完全无法触发**，是死代码。

## 三、GameViewModel 成就触发设计

### 3.1 对弈结束时触发点

在 `GameViewModel` 中新增成就检查方法：

```swift
// 新增文件：ViewModels/AchievementChecker.swift

/// 对弈结果信息（传给成就检查器）
struct GameResultInfo {
    let isWin: Bool
    let difficulty: AIDifficulty
    let moveCount: Int              // 总走法数（双方）
    let playerMoveCount: Int        // 玩家方走法数
    let elapsedSeconds: TimeInterval // 对局耗时
    let usedHint: Bool              // 是否用过提示
    let checkmatePattern: CheckmatePattern?  // 杀法类型（如有）
    let maxMaterialDeficit: Int     // 最大兵力劣势（负数=劣势，P1-1 补充详见 §3.5）
}

/// 杀法类型识别（简化版）
enum CheckmatePattern {
    case horseCannon    // 马后炮
    case doubleRook     // 双车错
    case other          // 其他
}

/// 成就检查器
enum AchievementChecker {
    
    /// 对弈结束后调用
    static func checkAfterGame(result: GameResultInfo, profile: PlayerProfile) -> [String] {
        guard result.isWin else { return [] }
        
        var unlocked: [String] = []
        let beaten = Set(profile.unlockedAchievements)
        
        // beat_medium / beat_hard / beat_master
        switch result.difficulty {
        case .medium: if !beaten.contains("beat_medium") { unlocked.append("beat_medium") }
        case .hard: if !beaten.contains("beat_hard") { unlocked.append("beat_hard") }
        case .master: if !beaten.contains("beat_master") { unlocked.append("beat_master") }
        default: break
        }
        
        // no_hint_win (P0-2 fix: 用 .order 而非 .rawValue)
        if !result.usedHint && result.difficulty.order >= AIDifficulty.medium.order
            && !beaten.contains("no_hint_win") {
            unlocked.append("no_hint_win")
        }
        
        // kill_ten_steps
        if result.playerMoveCount <= 10 && result.difficulty.order >= AIDifficulty.medium.order
            && !beaten.contains("kill_ten_steps") {
            unlocked.append("kill_ten_steps")
        }
        
        // blitz_5min
        if result.elapsedSeconds <= 300 && !beaten.contains("blitz_5min") {
            unlocked.append("blitz_5min")
        }
        
        // comeback_king
        if result.maxMaterialDeficit <= -5 && !beaten.contains("comeback_king") {
            unlocked.append("comeback_king")
        }
        
        // kill_mate_horse_cannon / kill_mate_double_rook
        if let pattern = result.checkmatePattern {
            switch pattern {
            case .horseCannon: if !beaten.contains("kill_mate_horse_cannon") { unlocked.append("kill_mate_horse_cannon") }
            case .doubleRook: if !beaten.contains("kill_mate_double_rook") { unlocked.append("kill_mate_double_rook") }
            case .other: break
            }
        }
        
        // first_blood (P1-4 fix: 重定义为“开局 5 步内将杀”)
        if result.playerMoveCount <= 5 && result.isWin
            && !beaten.contains("first_blood") {
            unlocked.append("first_blood")
        }
        
        // 连胜成就
        let currentStreak = profile.currentWinStreak + 1  // 本次胜利后 +1
        if currentStreak >= 5 && !beaten.contains("win_streak_5") { unlocked.append("win_streak_5") }
        if currentStreak >= 10 && !beaten.contains("win_streak_10") { unlocked.append("win_streak_10") }
        
        // all_difficulties (P0-3 fix: 5种难度，不是4种；排除 beginner)
        var beatenDiff = profile.beatenDifficulties
        beatenDiff.insert(result.difficulty.id)
        // 需要赢 easy/medium/hard/master 4种（排除 beginner）
        let requiredCount = AIDifficulty.allCases.filter { $0 != .beginner }.count
        if beatenDiff.count >= requiredCount && !beaten.contains("all_difficulties") {
            unlocked.append("all_difficulties")
        }
        
        return unlocked
    }
    
    /// 残局通关后调用
    static func checkAfterPuzzle(puzzleType: String, chapterId: String?, 
                                  allPuzzleCount: Int, completedCount: Int,
                                  profile: PlayerProfile) -> [String] {
        var unlocked: [String] = []
        let beaten = Set(profile.unlockedAchievements)
        
        // first_draw_puzzle
        if puzzleType == "draw" && !beaten.contains("first_draw_puzzle") {
            unlocked.append("first_draw_puzzle")
        }
        
        // chapter1_clear (P1-3 fix: 明确触发链路)
        if chapterId == "ch1" && !beaten.contains("chapter1_clear") {
            // 直接查 ChapterStore
            if let ch1 = ChapterStore.shared.chapter(byId: "ch1"), ch1.isComplete {
                unlocked.append("chapter1_clear")
            }
        }
        
        // endgame_master（50局残局通关）
        if completedCount >= 50 && !beaten.contains("endgame_master") {
            unlocked.append("endgame_master")
        }
        
        // all_puzzles（551局全通关）
        if completedCount >= allPuzzleCount && !beaten.contains("all_puzzles") {
            unlocked.append("all_puzzles")
        }
        
        return unlocked
    }
}
```

### 3.2 GameViewModel 集成

```swift
// GameViewModel.swift 新增
func recordWin() {
    // 现有逻辑：更新 totalWins...
    
    // 成就检查
    let result = GameResultInfo(
        isWin: true,
        difficulty: currentDifficulty,
        moveCount: gameMoves.count,
        playerMoveCount: gameMoves.filter { $0.piece.side == playerSide }.count,
        elapsedSeconds: gameStartTime.distance(to: Date()),
        usedHint: usedHintFlag,
        checkmatePattern: detectCheckmatePattern(),
        maxMaterialDeficit: materialTracker.maxDeficit
    )
    
    let newAchievements = AchievementChecker.checkAfterGame(
        result: result, profile: PlayerProfileStore.shared.profile
    )
    for id in newAchievements {
        AchievementManager.shared.unlock(id)
    }
    
    // Q3: 成就解锁后触发段位检查
    if !newAchievements.isEmpty {
        checkRankPromotion()
    }
}

/// 检测杀法类型（简化版）
private func detectCheckmatePattern() -> CheckmatePattern? {
    // 看最后几步的棋子组合
    let lastMoves = suffix(5)  // 最后 5 步
    let playerPieces = lastMoves.filter { $0.piece.side == playerSide }
    
    let hasHorse = playerPieces.contains { $0.piece.kind == .horse }
    let hasCannon = playerPieces.contains { $0.piece.kind == .cannon }
    let hasRook = playerPieces.contains { $0.piece.kind == .chariot }
    
    if hasHorse && hasCannon { return .horseCannon }
    let rookCount = playerPieces.filter { $0.piece.kind == .chariot }.count
    if rookCount >= 2 { return .doubleRook }
    return .other
}
```

### 3.3 连胜追踪

当前 `PlayerProfile` 没有 `currentWinStreak` 字段。需要新增：

```swift
// PlayerProfile 新增
var currentWinStreak: Int = 0
var maxWinStreak: Int = 0
var beatenDifficulties: [String] = []  // 已击败的难度 ID 列表

// GameViewModel 中：
func recordResult(_ result: GameResult) {
    PlayerProfileStore.shared.update { profile in
        switch result {
        case .win:
            profile.totalWins += 1
            profile.currentWinStreak += 1
            if profile.currentWinStreak > profile.maxWinStreak {
                profile.maxWinStreak = profile.currentWinStreak
            }
            // 记录击败的难度
            if !profile.beatenDifficulties.contains(currentDifficulty.id) {
                profile.beatenDifficulties.append(currentDifficulty.id)
            }
        case .lose:
            profile.totalLosses += 1
            profile.currentWinStreak = 0  // 重置连胜
        case .draw:
            profile.totalDraws += 1
            // P1-3 fix: 和局不影响 currentWinStreak（不加不减）
        }
    }
}
```

### 3.5 MaterialTracker 设计（P1-1 补充）

`comeback_king` 成就需要追踪对局中的最大兵力劣势。设计如下：

```swift
/// 子力差追踪器（在 GameViewModel 中使用）
struct MaterialTracker {
    private(set) var maxDeficit: Int = 0  // 最大劣势（负数）
    private let pieceValues: [PieceKind: Int] = [
        .general: 100, .chariot: 9, .horse: 4, .cannon: 5,
        .advisor: 2, .elephant: 2, .soldier: 1
    ]

    /// 每步走完后调用
    mutating func update(board: Board, playerSide: Side) {
        let redValue = board.pieces.filter { $0.side == .red }
            .reduce(0) { $0 + (pieceValues[$1.kind] ?? 0) }
        let blackValue = board.pieces.filter { $0.side == .black }
            .reduce(0) { $0 + (pieceValues[$1.kind] ?? 0) }
        let diff = playerSide == .red ? redValue - blackValue : blackValue - redValue
        if diff < maxDeficit {
            maxDeficit = diff
        }
    }
}
```

## 四、成就→段位联动

### 4.1 effectiveWins/effectivePuzzles 统一计算（P0-1 修正）

**P0-1 问题**：如果只改 `checkRankUp()` 而不改 `canPromote(to:)`，两个方法用不同判断标准，导致进度条显示与实际升级矛盾。

**修复**：把 achievementBonus 逻辑统一到计算属性，`canPromote` 和 `checkRankUp` 都引用它：

```swift
// PlayerProfile 新增计算属性
var effectiveWins: Int { totalWins + achievementBonusWins }
var effectivePuzzles: Int { puzzlesCompleted + achievementBonusPuzzles }

/// 成就贡献的虚拟胜场数（P1-2 修正：降低数值，总上限 ~9）
private var achievementBonusWins: Int {
    var bonus = 0
    if unlockedAchievements.contains("tutorial_done") { bonus += 1 }
    if unlockedAchievements.contains("beat_medium") { bonus += 1 }   // 降为 +1
    if unlockedAchievements.contains("beat_hard") { bonus += 2 }     // 降为 +2
    if unlockedAchievements.contains("beat_master") { bonus += 3 }   // 降为 +3
    return bonus  // 最大 +7
}

/// 成就贡献的虚拟残局通关数（P1-2 修正）
private var achievementBonusPuzzles: Int {
    var bonus = 0
    if unlockedAchievements.contains("first_puzzle") { bonus += 1 }
    if unlockedAchievements.contains("chapter1_clear") { bonus += 3 }     // 降为 +3
    if unlockedAchievements.contains("endgame_master") { bonus += 5 }     // 降为 +5
    return bonus  // 最大 +9
}

// canPromote 修改为用 effective 值
func canPromote(to target: Rank) -> Bool {
    effectiveWins >= target.requiredWins && effectivePuzzles >= target.requiredPuzzles
}

// checkRankUp 不变（已通过 canPromote 引用 effective 值）
func checkRankUp() -> Rank? {
    var bestRank = rank
    for candidate in Rank.allCases {
        if candidate > bestRank && canPromote(to: candidate) {
            bestRank = candidate
        }
    }
    return bestRank > rank ? bestRank : nil
}
```

**进度条显示**：RankProgressView 中显示 effective 值（包含成就加成）：

```swift
Text("\(L10n.shared.t("achievement.view.wins"))：\(profile.effectiveWins)/\(target.requiredWins)")
```

### 4.2 段位升级触发

在 `AchievementManager.unlock()` 中增加段位检查：

```swift
@discardableResult
func unlock(_ achievementId: String) -> Bool {
    let current = store.profile
    if current.unlockedAchievements.contains(achievementId) { return false }
    
    _ = store.update { profile in
        profile.unlockedAchievements.append(achievementId)
    }
    
    // Q3: 成就可能触发段位升级
    checkRankPromotion()
    
    return true
}

private func checkRankPromotion() {
    let profile = store.profile
    if let newRank = profile.checkRankUp() {
        store.update { $0.rank = newRank }
        NotificationCenter.default.post(
            name: .rankPromoted, object: newRank
        )
    }
}
```

## 五、成就解锁奖励（修订版）

P2-3 修正：gold 成就奖励从"解锁 3★ 残局提前访问"改为"额外提示次数"，避免与章节系统耦合。

| 稀有度 | 奖励 | 实现方式 |
|--------|------|---------|
| bronze | 段位贡献（见 §4.1 achievementBonus） | checkRankUp 中加成 |
| silver | 每日额外提示 +1/天 | PlayerProfile 新增 `bonusExtraHints: Int` |
| gold | 每日额外提示 +2/天（累计） | 同上 |
| diamond | 解锁专属棋子样式 | PlayerProfile 新增 `bonusPieceStyle: Bool`（已有此字段） |
| hidden | 解锁特殊效果 | PlayerProfile `bonusSpecialTheme: Bool`（已有） |

**silver/gold 的额外提示**（P2-1 修正：设上限）

```swift
// PlayerProfile
var bonusExtraHints: Int = 0  // 每日额外提示次数（由成就解锁增加）

// 计算总提示次数（上限 10）
var totalDailyHints: Int {
    let base = 3  // 基础 3 次/天
    return min(base + bonusExtraHints, 10)  // P2-1 fix: 上限 10 次/天
}

// AchievementManager.unlock() 后根据稀有度加提示
private func applyUnlockReward(_ achievement: Achievement) {
    switch achievement.rarity {
    case .silver:
        store.update { $0.bonusExtraHints += 1 }
    case .gold:
        store.update { $0.bonusExtraHints += 2 }
    case .diamond:
        store.update { $0.bonusPieceStyle = true }
    case .hidden:
        store.update { $0.bonusSpecialTheme = true }
    case .bronze:
        break  // 段位贡献已通过 effectiveWins 处理
    }
}
```
