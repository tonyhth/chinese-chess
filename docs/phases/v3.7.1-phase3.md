#### 1.4.1 PGNImporter 扩展

```swift
// PGNImporter.swift — parse() 返回值扩展

struct ImportResult {
    let records: [GameRecord]
    let warnings: [String]
    let skippedCount: Int    // 新增：跳过的局数
    let totalGames: Int      // 新增：检测到的总局数
    
    var isSuccess: Bool { !records.isEmpty }
    var hasWarnings: Bool { !warnings.isEmpty }
}
```

**设计决策**（Vera P0-1）：不提供 `init(records:warnings:)` 便利构造器。`skippedCount` 必须从 parse 的实际逻辑推算（总局数 - 成功数），而非硬绑 `warnings.count`。现有 `parse()` 是唯一构造点，直接使用 full init。

`parse()` 方法内部计算 `skippedCount` 和 `totalGames`：

```swift
static func parse(_ pgnText: String) -> ImportResult {
    var records: [GameRecord] = []
    var warnings: [String] = []
    let games = splitGames(pgnText)
    let totalGames = games.count

    for (i, gameText) in games.enumerated() {
        do {
            let record = try parseSingleGame(gameText)
            records.append(record)
        } catch {
            warnings.append("第 \(i + 1) 局解析失败：\(error.localizedDescription)")
        }
    }

    return ImportResult(
        records: records,
        warnings: warnings,
        skippedCount: totalGames - records.count,
        totalGames: totalGames
    )
}
```

**兼容旧测试**：V370Phase3Tests 等已用 `ImportResult(records:warnings:)` 的测试需更新为 full init，在 `skippedCount` 传 `0`、`totalGames` 传 `records.count`。

#### 1.4.2 GameRecordStore.batchAdd

```swift
// GameRecordStore.swift

/// 批量添加记录，返回实际新增数量
/// 单次 writeQueue.sync 内完成所有写入，只调用一次 persistIndex()
@discardableResult
func batchAdd(_ records: [GameRecord]) -> Int {
    writeQueue.sync {
        var added = 0
        for record in records {
            // puzzleId 去重
            if let pid = record.puzzleId,
               summaries.contains(where: { $0.puzzleId == pid }) {
                continue
            }
            // 直接操作内存，不逐条调 addRecord
            summaries.append(RecordSummary(from: record))
            persistSingleRecord(record)
            added += 1
        }
        if added > 0 {
            persistIndex()
        }
        return added
    }
}
```

**设计决策**（Vera P1-6）：`batchAdd` 在单次 `writeQueue.sync` 内完成所有操作，只写一次 index.json。不循环调用 `addRecord()`（每条都写 index.json，性能差且中途崩溃导致部分入库无原子性）。

#### 1.4.3 ImportViewModel（新增）

```swift
// ViewModels/ImportViewModel.swift

import SwiftUI

@Observable
class ImportViewModel {
    enum ImportState {
        case idle           // 等待选择文件
        case parsing        // 解析中
        case success(ImportResult)  // 解析完成
        case failure(String)         // 解析失败
    }
    
    private(set) var state: ImportState = .idle
    
    /// 从 PGN 文本解析
    /// 解析在后台执行，结果回调回 MainActor
    func parse(pgnText: String) {
        state = .parsing
        Task.detached {
            // 解析在后台线程执行，不阻塞 UI
            let result = PGNImporter.parse(pgnText)
            await MainActor.run {
                if result.records.isEmpty && result.totalGames == 0 {
                    self.state = .failure("无法识别有效的 PGN 内容")
                } else {
                    self.state = .success(result)
                }
            }
        }
    }
    
    /// 确认导入，将记录写入 Store
    func confirmImport() -> Int {
        guard case .success(let result) = state else { return 0 }
        let added = GameRecordStore.shared.batchAdd(result.records)
        state = .idle
        return added
    }
    
    /// 重置
    func reset() {
        state = .idle
    }
}
```

**设计决策**（Vera P0-2）：使用 `Task.detached` 在后台线程执行解析，`MainActor.run` 回调结果。之前的 `Task { @MainActor in }` 只是延迟到下一个 MainActor hop，解析仍在主线程。

#### 2.4.1 PuzzleViewModel 通关自动存储

**核心设计**（Vera P1-4/P1-5）：使用 `gameState` 的 `didSet` 观察器触发存储，一处声明永不遗漏；直接使用 `buildSolutionRecord()` 的输出，不重新构建 record。

```swift
// PuzzleViewModel.swift — gameState didSet 观察器

var gameState: GameState = .playing {
    didSet {
        if gameState == .redWon || gameState == .blackWon {
            savePuzzleRecord()
        }
    }
}

/// 通关后自动保存棋谱记录到 GameRecordStore
private func savePuzzleRecord() {
    // 直接使用 buildSolutionRecord()，它已有正确的 moves、notation、isCheck/isCheckmate
    guard let baseRecord = buildSolutionRecord() else { return }
    
    // 补上 source 和 puzzleId，生成最终 record
    let record = GameRecord(
        id: baseRecord.id,
        title: baseRecord.title.isEmpty ? puzzle.name : baseRecord.title,
        date: Date(),
        redPlayer: baseRecord.redPlayer,
        blackPlayer: baseRecord.blackPlayer,
        difficulty: baseRecord.difficulty,
        result: baseRecord.result,
        totalMoves: baseRecord.totalMoves,
        moves: baseRecord.moves,
        initialFEN: baseRecord.initialFEN,
        source: .puzzle,
        puzzleId: puzzle.id
    )
    
    // A4 去重：同 puzzleId 覆盖旧记录
    if let existing = GameRecordStore.shared.findRecordByPuzzleId(puzzle.id) {
        let preserved = GameRecord(
            id: existing.id,
            title: record.title,
            date: record.date,
            redPlayer: record.redPlayer,
            blackPlayer: record.blackPlayer,
            difficulty: record.difficulty,
            result: record.result,
            totalMoves: record.totalMoves,
            moves: record.moves,
            initialFEN: record.initialFEN,
            source: .puzzle,
            puzzleId: puzzle.id
        )
        GameRecordStore.shared.updateRecord(preserved)
    } else {
        GameRecordStore.shared.addRecord(record)
    }
}
```

**为什么用 didSet 而不是手动追加调用**：

`gameState = .redWon` / `.blackWon` 在 PuzzleViewModel 中有至少 4 处设置点：
1. `movePiece()` — freePlay 将死对方
2. `movePiece()` — sequence 局走完推荐步数
3. `handleCorrectMove()` — guided 将死对方
4. `handleCorrectMove()` — guided 走完 solution

用 `didSet` 一处声明，不会遗漏任何通关路径。

**为什么用 `buildSolutionRecord()` 而不重新构建**：

`buildSolutionRecord()` 已经包含正确的 moves、notation、isCheck/isCheckmate。重新构建的 fallback 构造器 result 硬编码为 `puzzle.side == .red ? .redWon : .blackWon`，可能与 `buildSolutionRecord()` 的实际 result 不一致（Vera P1-5）。

## 六、注意事项

1. **RecordSummary 无需修改**（Vera P2-9）：`RecordSummary` 已包含 `source` 和 `totalMoves` 字段（第 15、14 行），Phase 3C 无需改动。

2. **ImportResult 不提供便利构造器**（Vera P0-1）：旧 `init(records:warnings:)` 的 `skippedCount` 语义有误。V370Phase3Tests 等旧测试需更新为 full init。

3. **PuzzleViewModel.savePuzzleRecord 用 didSet 触发**（Vera P1-4）：`gameState` 有至少 4 处设置 `.redWon/.blackWon` 的位置，用 `didSet` 一处声明，不会遗漏。

4. **fileImporter macOS 适配**：macOS 上 `fileImporter` 行为与 iOS 不同（不弹出 sheet），需要额外处理。可考虑 macOS 使用 `NSOpenPanel` 或条件编译。

5. **RecordPanelView 修复两步必须都做**（Vera P0-3）：第一步删除 `moveText()` 内部导出代码，第二步在 `body` 底部添加操作栏。只做第二步会导致两个导出按钮同时出现。

6. **UTType("pgn") 可能不存在**（Vera P1-7）：iOS 上 `.pgn` 扩展名可能未注册 UTType。`fileImporter` 的 `allowedContentTypes` 需 fallback 到 `UTType.plainText`，解析阶段再校验内容。

7. **ImportViewModel 用 Task.detached 后台解析**（Vera P0-2）：`Task { @MainActor in }` 只是延迟到下一个 MainActor hop，解析仍在主线程。必须用 `Task.detached` + `MainActor.run` 回调。
